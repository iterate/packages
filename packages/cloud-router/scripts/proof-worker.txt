import { IterateConfigEntrypoint, type IterateConfigProcessEventArgs } from "iterate/sdk";

// A test project on iterate-nov7 for the Cloud's router (iterate/private packages/cloud-router): its
// home page names the project and the routing slug a host carries (blog--<slug>.iterate2.app), and
// x--<slug>.iterate2.app/?path=/2/users/me/bookmarks reads the owner's X through the Cloud;
// ask--<slug>.iterate2.app/?q=<question> asks a model through the Cloud, with a key the Cloud lends this deployment and
// this project never holds: the request goes to the router, which fills `cloudSecret("openai")`.
export default class extends IterateConfigEntrypoint {
  // Every durable event of the project, at least once. A message to @iterate from the owner's X
  // account lands on /iterate-cloud/integrations/x/<account> (the Cloud's router); this project
  // answers it with the model through the Cloud, as an ask the router sends as @iterate.
  async processEvent({ event, itx }: IterateConfigProcessEventArgs) {
    if (event.type !== "x/message-received") return;
    const message = event.payload as { kind: "dm" | "mention"; id: string; text: string; from: { xUserId: string; username: string } };
    const answer = await fetch("https://iterate2.app/.cloud/egress/api.openai.com/v1/chat/completions", {
      method: "POST",
      headers: {
        "content-type": "application/json",
        "x-itx-cloud": 'getSecret("/secrets/iterate-cloud")',
        authorization: 'Bearer cloudSecret("openai")',
      },
      body: JSON.stringify({
        model: "gpt-6-luna",
        max_completion_tokens: message.kind === "dm" ? 400 : 120,
        messages: [
          { role: "system", content: `You are @iterate, an assistant a person reached on X. Answer in at most ${message.kind === "dm" ? "four sentences" : "240 characters"}, plainly, no hashtags.` },
          { role: "user", content: message.text },
        ],
      }),
    });
    const body = await answer.text();
    let text = `I could not think of an answer (HTTP ${answer.status}).`;
    try {
      const parsed = JSON.parse(body) as { choices?: { message?: { content?: string } }[] };
      text = parsed.choices?.[0]?.message?.content?.trim() || text;
    } catch {}
    await itx.cd(event.path).append({
      type: "x/reply-requested",
      idempotencyKey: `x/reply:${message.kind}:${message.id}`,
      payload: message.kind === "dm"
        ? { kind: "dm", to: message.from.xUserId, text }
        : { kind: "mention", inReplyTo: message.id, text: `@${message.from.username} ${text}`.slice(0, 280) },
    });
  }

  async fetch(request: Request) {
    const url = new URL(request.url);
    using itx = this.getItx();
    const { projectSlug } = await itx.whoami();
    const routingSlug = request.headers.get("x-iterate-routing-slug");
    if (routingSlug === "x") {
      // the owner's X account, connected at the Cloud: the Cloud makes the request, this project holds no token
      const path = url.searchParams.get("path") || "/2/users/me";
      const answer = await fetch(`https://iterate2.app/.cloud/egress/api.x.com${path}`, {
        headers: { "x-itx-cloud": 'getSecret("/secrets/iterate-cloud")', authorization: 'Bearer cloudSecret("x")' },
      });
      return new Response(`${await answer.text()}\n\n(${answer.status} from api.x.com through iterate2.app; project ${projectSlug} holds no token)\n`, {
        status: answer.status,
        headers: { "content-type": "text/plain; charset=utf-8" },
      });
    }
    if (routingSlug === "ask") {
      const question = url.searchParams.get("q") || "Say hello in five words.";
      const started = Date.now();
      const answer = await fetch("https://iterate2.app/.cloud/egress/api.openai.com/v1/chat/completions", {
        method: "POST",
        headers: {
          "content-type": "application/json",
          // the deployment's own egress fills this from the project's secret, pinned to the router
          "x-itx-cloud": 'getSecret("/secrets/iterate-cloud")',
          // the router fills this with the key the Cloud lends the deployment
          authorization: 'Bearer cloudSecret("openai")',
        },
        body: JSON.stringify({ model: "gpt-6-luna", messages: [{ role: "user", content: question }], max_completion_tokens: 200 }),
      });
      const body = await answer.text();
      let reply = body;
      try {
        const parsed = JSON.parse(body);
        reply = parsed.choices?.[0]?.message?.content ?? body;
      } catch {}
      return new Response(`${reply}\n\n(${answer.status} from api.openai.com through iterate2.app in ${Date.now() - started} ms; project ${projectSlug} holds no key)\n`, {
        status: answer.ok ? 200 : 502,
        headers: { "content-type": "text/plain; charset=utf-8" },
      });
    }
    const lines = [
      routingSlug ? `The ${routingSlug} of project ${projectSlug}` : `Homepage of project ${projectSlug}`,
      `path ${url.pathname}${url.search}`,
    ];
    return new Response(lines.join("\n") + "\n", {
      headers: { "content-type": "text/plain; charset=utf-8" },
    });
  }
}
